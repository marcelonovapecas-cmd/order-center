schema:

enum Cargos {
    EMPREGADO
    GERENTE
    ADMIN
}

model Usuario {
    id          Int         @id @default(autoincrement())
    nome        String  
    email       String      @unique
    senha       String
    cargo       Cargos      @default(EMPREGADO)      
    createdAt   DateTime    @default(now())

    pedidos     Pedido[]
    eventos     EventoPedido[]
}

enum Status {
    AGUARDANDO
    EM_SEPARACAO
    CONCLUIDO
    CANCELADO
}

model Pedido {
    id          Int         @id @default(autoincrement())
    numero      Int         @unique @default(autoincrement())
    status      Status      @default(AGUARDANDO)

    usuarioId   Int
    usuario     Usuario     @relations(fields: [usuarioId], references:[id])

    itens       ItemPedido[]
    eventos     EventoPedido[]   
    
    createdAt   DateTime    @default(now())
    updatedAt   DateTime    @updatedAt
}

model ItemPedido {
    id          Int         @id @default(autoincrement())

    pedidoId    Int
    pedido      Pedido      @relations(fields:[pedidoId],references:[id])
    
    nome        String
    quantidade  Int
    createdAt   DateTime    @default(now())
}

enum TiposEvento {
    PEDIDO_CRIADO
    STATUS_ALTERADO
    ITEM_ADICIONADO
    ITEM_REMOVIDO
    QUANTIDADE_ALTERADA
}

model EventoPedido {
    id          Int             @id @default(autoincrement())
    
    pedidoId    Int    
    pedido      Pedido          @relations(fields:[pedidoId], references:[id])

    usuarioId   Int
    usuario     Usuario         @relations(fields:[usuarioId],references:[id])
    
    tipo        TiposEvento    
    descricao   String
    createdAt   DateTime        @default(now())
}
